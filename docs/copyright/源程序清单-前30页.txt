   170  //! action_router：五动作统一入口与全链路汇聚（T16 完整实现；spec 5.4/5.6/7.2、FR-11.4、5.7.3）。
   171  //!
   172  //! 依赖方向：action_router → 各能力模块（selection/clipboard_monitor/ocr/capture/translate/windows）

   175  //! 汇聚入口（全部经 `dispatch` 或本模块命令）：

   177  //! - 剪贴板监听（ClipboardTranslateTrigger 注入）→ `clipboard_translate`（挂起/门控同径）；
   178  //! - 框选确认延续（mask `SelectionCompletedHandler` 注入）→ `continue_ocr_flow`（识别/送翻分路）；
   179  //! - 面板提交（`translate` / `send_ocr_to_translate` 命令）→ 本模块门控后进 service（EX-13 单路径）。
   180  //!

   183  //! 2. `hasKey==false`（翻译类入口）→ 不发起网络请求、打开引导窗 guide 模式、动作挂起（FR-3.4/4.4/7.3）；

  1416  /// 阻塞等待 IAsyncOperation（与 ocr::wait_async 同构；本模块私有精简版）。
  1417  fn wait_async<T>(operation: windows_future::IAsyncOperation<T>) -> Result<T, windows::core::Error>
  1418  where
  1419      T: windows::core::RuntimeType + Send + 'static,
  1420  {
  1421      use windows_future::{AsyncOperationCompletedHandler, AsyncStatus};
  1422      use std::sync::mpsc;
  1423      match operation.Status()? {
  1424          AsyncStatus::Completed => return operation.GetResults(),
  1425          AsyncStatus::Error => {
  1426              return Err(windows::core::Error::from_hresult(operation.ErrorCode()?))
  1427          }
  1428          _ => {}
  1429      }
  1430      let (tx, rx) = mpsc::channel::<Result<T, windows::core::Error>>();
  1431      operation.SetCompleted(&AsyncOperationCompletedHandler::new(move |sender, status| {
  1432          let op = sender.ok()?;
  1433          let result = match status {
  1434              AsyncStatus::Completed => op.GetResults(),
  1435              AsyncStatus::Error => match op.ErrorCode() {
  1436                  Ok(code) => Err(windows::core::Error::from_hresult(code)),
  1437                  Err(e) => Err(e),
  1438              },
  1439              _ => Err(windows::core::Error::from_hresult(
  1440                  windows::core::HRESULT::from_win32(1223), // ERROR_CANCELLED
  1441              )),
  1442          };
  1443          let _ = tx.send(result);
  1444          Ok(())
  1445      }))?;
  1446      rx.recv().map_err(|_| {
  1447          windows::core::Error::from_hresult(windows::core::HRESULT::from_win32(1223))
  1448      })?
  1449  }

  1465  fn apply_packaged(target_on: bool) -> Result<(), AppError> {
  1466      use windows::ApplicationModel::StartupTaskState;
  1467      let lang = crate::ui_text::resolve_global();
  1468      let task = packaged_startup_task()?;
  1469      let state = task
  1470          .State()
  1471          .map_err(|err| AppError::Config(lang.autostart_query_failed(&err.to_string())))?;
  1472      let currently_enabled =
  1473          matches!(state, StartupTaskState::Enabled | StartupTaskState::EnabledByPolicy);
  1474      match decide_action(target_on, currently_enabled) {
  1475          AutostartAction::Enable => {
  1476              if currently_enabled {
  1477                  log::debug!("autostart: 包身份目标开启且已启用（幂等）");
  1478                  return Ok(());
  1479              }
  1480              match state {
  1481                  StartupTaskState::Disabled => {
  1482                      let enable_op = task
  1483                          .RequestEnableAsync()
  1484                          .map_err(|err| {
  1485                              AppError::Config(lang.autostart_write_failed(&err.to_string()))
  1486                          })?;
  1487                      let new_state = wait_async(enable_op).map_err(|err| {
  1488                          AppError::Config(lang.autostart_write_failed(&err.to_string()))
  1489                      })?;
  1490                      if !matches!(
  1491                          new_state,
  1492                          StartupTaskState::Enabled | StartupTaskState::EnabledByPolicy
  1493                      ) {
  1494                          return Err(AppError::Config(lang.autostart_enable_rejected()));
  1495                      }
  1496                  }
  1497                  StartupTaskState::DisabledByUser => {
