   172  //!

   176  //! 汇聚入口（全部经 `dispatch` 或本模块命令）：

   178  //! - 剪贴板监听（ClipboardTranslateTrigger 注入）→ `clipboard_translate`（挂起/门控同径）；
   179  //! - 框选确认延续（mask `SelectionCompletedHandler` 注入）→ `continue_ocr_flow`（识别/送翻分路）；
   180  //! - 面板提交（`translate` / `send_ocr_to_translate` 命令）→ 本模块门控后进 service（EX-13 单路径）。

   184  //! 2. `hasKey==false`（翻译类入口）→ 不发起网络请求、打开引导窗 guide 模式、动作挂起（FR-3.4/4.4/7.3）；

  1417  /// 阻塞等待 IAsyncOperation（与 ocr::wait_async 同构；本模块私有精简版）。
  1418  fn wait_async<T>(operation: windows_future::IAsyncOperation<T>) -> Result<T, windows::core::Error>
  1419  where
  1420      T: windows::core::RuntimeType + Send + 'static,
  1421  {
  1422      use windows_future::{AsyncOperationCompletedHandler, AsyncStatus};
  1423      use std::sync::mpsc;
  1424      match operation.Status()? {
  1425          AsyncStatus::Completed => return operation.GetResults(),
  1426          AsyncStatus::Error => {
  1427              return Err(windows::core::Error::from_hresult(operation.ErrorCode()?))
  1428          }
  1429          _ => {}
  1430      }
  1431      let (tx, rx) = mpsc::channel::<Result<T, windows::core::Error>>();
  1432      operation.SetCompleted(&AsyncOperationCompletedHandler::new(move |sender, status| {
  1433          let op = sender.ok()?;
  1434          let result = match status {
  1435              AsyncStatus::Completed => op.GetResults(),
  1436              AsyncStatus::Error => match op.ErrorCode() {
  1437                  Ok(code) => Err(windows::core::Error::from_hresult(code)),
  1438                  Err(e) => Err(e),
  1439              },
  1440              _ => Err(windows::core::Error::from_hresult(
  1441                  windows::core::HRESULT::from_win32(1223), // ERROR_CANCELLED
  1442              )),
  1443          };
  1444          let _ = tx.send(result);
  1445          Ok(())
  1446      }))?;
  1447      rx.recv().map_err(|_| {
  1448          windows::core::Error::from_hresult(windows::core::HRESULT::from_win32(1223))
  1449      })?
  1450  }

  1466  fn apply_packaged(target_on: bool) -> Result<(), AppError> {
  1467      use windows::ApplicationModel::StartupTaskState;
  1468      let lang = crate::ui_text::resolve_global();
  1469      let task = packaged_startup_task()?;
  1470      let state = task
  1471          .State()
  1472          .map_err(|err| AppError::Config(lang.autostart_query_failed(&err.to_string())))?;
  1473      let currently_enabled =
  1474          matches!(state, StartupTaskState::Enabled | StartupTaskState::EnabledByPolicy);
  1475      match decide_action(target_on, currently_enabled) {
  1476          AutostartAction::Enable => {
  1477              if currently_enabled {
  1478                  log::debug!("autostart: 包身份目标开启且已启用（幂等）");
  1479                  return Ok(());
  1480              }
  1481              match state {
  1482                  StartupTaskState::Disabled => {
  1483                      let enable_op = task
  1484                          .RequestEnableAsync()
  1485                          .map_err(|err| {
  1486                              AppError::Config(lang.autostart_write_failed(&err.to_string()))
  1487                          })?;

  1489                          AppError::Config(lang.autostart_write_failed(&err.to_string()))
  1490                      })?;

  1492                          new_state,
  1493                          StartupTaskState::Enabled | StartupTaskState::EnabledByPolicy

  1495                          return Err(AppError::Config(lang.autostart_enable_rejected()));
